Simple Integration

Get up and running in minutes with our intuitive JavaScript library

meta-exec-lib Example
JavaScript
import { buildCallData, prepareForward, signForward, executeForward } from "meta-exec-lib";

// 1. Build the calldata for your target contract
const callData = buildCallData(
  ["function store(uint256)"], 
  "store", 
  [valueToStore]
);

// 2. Prepare the meta-transaction
const prep = await prepareForward({
  provider,
  metaAddress: HUB_ADDRESS,
  from: user.address,
  to: STORAGE_ADDRESS,
  callData,
  caller: relayer.address,
  nonce: Math.floor(Math.random() * 1000000)
});

// 3. User signs the transaction
const signature = await signForward(user, prep.domain, prep.types, prep.message);

// 4. Relayer executes on-chain
const tx = await executeForward({
  provider,
  metaAddress: HUB_ADDRESS,
  fTuple: prep.fTuple,
  signature,
  relayer
});

console.log("✅ Transaction executed:", tx.hash);
